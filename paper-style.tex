% ACM-like single-column style for paper.md (included by build_paper.sh).
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{etoolbox}
\usepackage{caption}

% Fonts: Libertine text, Biolinum headings, Libertinus maths, Inconsolata code
\setsansfont{Linux Biolinum O}
\setmonofont[Scale=MatchLowercase]{Inconsolatazi4-Regular.otf}
\setmathfont{LibertinusMath-Regular.otf}
\AtBeginDocument{\renewcommand{\boldsymbol}[1]{\symbf{#1}}}

% ACM link colours
\definecolor{ACMPurple}{cmyk}{0.55,1,0,0.15}
\definecolor{ACMDarkBlue}{cmyk}{1,0.58,0,0.21}
\AtBeginDocument{\hypersetup{colorlinks=true, linkcolor=ACMDarkBlue,
  citecolor=ACMPurple, urlcolor=ACMDarkBlue}}

% Headings: numbers are written in the headings themselves
\titleformat{\section}{\normalfont\sffamily\bfseries\normalsize}{}{0pt}{\MakeUppercase}
\titleformat{\subsection}{\normalfont\sffamily\bfseries\normalsize}{}{0pt}{}
\titleformat{\subsubsection}[runin]{\normalfont\sffamily\itshape\normalsize}{}{0pt}{}[.]
\titlespacing*{\section}{0pt}{14pt plus 2pt minus 2pt}{6pt}
\titlespacing*{\subsection}{0pt}{10pt plus 2pt minus 2pt}{4pt}
\titlespacing*{\subsubsection}{0pt}{8pt plus 2pt}{0.5em}

% Title block: flush left, sans serif
\makeatletter
\renewcommand{\@maketitle}{%
  \begin{flushleft}
    {\sffamily\bfseries\LARGE\@title\par}\vskip 1.2em
    {\sffamily\large\def\and{\unskip, }\@author\par}\vskip 0.4em
    {\sffamily\small\@date\par}
  \end{flushleft}\vskip 1em}
\makeatother

% Abstract: small, with a rule underneath
\renewenvironment{abstract}
  {\small\noindent\ignorespaces}
  {\par\vskip 0.8em\noindent\rule{\linewidth}{0.4pt}\par\vskip 1em}

\pagestyle{fancy}

\fancyhead[R]{\sffamily\footnotesize\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}}
